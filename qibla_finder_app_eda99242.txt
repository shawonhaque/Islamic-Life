import React, { useState, useEffect, useRef } from 'react';
import { Info, Compass, MapPin, Check, X } from 'lucide-react';

// Kaaba Coordinates (Makkah)
const KAABA_LAT = 21.4225;
const KAABA_LON = 39.8262;

// Default Fallback Coordinates (Dhaka, Bangladesh)
const DEFAULT_COORDS = {
  lat: 23.8103,
  lon: 90.4125,
  city: 'Dhaka',
};

function toRadians(deg: number) {
  return (deg * Math.PI) / 180;
}

function toDegrees(rad: number) {
  return (rad * 180) / Math.PI;
}

// Calculate Qibla Bearing & Distance using Spherical Trigonometry
function calculateQiblaData(lat: number, lon: number) {
  const phi1 = toRadians(lat);
  const phi2 = toRadians(KAABA_LAT);
  const deltaLambda = toRadians(KAABA_LON - lon);

  const y = Math.sin(deltaLambda) * Math.cos(phi2);
  const x =
    Math.cos(phi1) * Math.sin(phi2) -
    Math.sin(phi1) * Math.cos(phi2) * Math.cos(deltaLambda);

  let bearing = toDegrees(Math.atan2(y, x));
  bearing = (bearing + 360) % 360;

  // Haversine Distance in km
  const R = 6371;
  const dLat = phi2 - phi1;
  const a =
    Math.sin(dLat / 2) * Math.sin(dLat / 2) +
    Math.cos(phi1) *
      Math.cos(phi2) *
      Math.sin(deltaLambda / 2) *
      Math.sin(deltaLambda / 2);
  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
  const distanceKm = Math.round(R * c);
  const distanceMi = Math.round(distanceKm * 0.621371);

  return {
    bearing: Number(bearing.toFixed(1)),
    distanceKm,
    distanceMi,
  };
}

function getCardinalDirection(deg: number) {
  const dirs = ['N', 'NE', 'E', 'SE', 'S', 'SW', 'W', 'NW'];
  const index = Math.round(deg / 45) % 8;
  return dirs[index];
}

export default function App() {
  const [showTutorial, setShowTutorial] = useState(true);
  const [location, setLocation] = useState(DEFAULT_COORDS);
  const [qiblaData, setQiblaData] = useState(() =>
    calculateQiblaData(DEFAULT_COORDS.lat, DEFAULT_COORDS.lon)
  );

  // Real-time Compass States
  const [heading, setHeading] = useState<number | null>(null);
  const [needsPermission, setNeedsPermission] = useState(false);
  const [mockDemoAngle, setMockDemoAngle] = useState(62);
  const hasBuzzedRef = useRef(false);

  // Animate the tutorial phone mockup arrow smoothly
  useEffect(() => {
    const interval = setInterval(() => {
      setMockDemoAngle((prev) => (prev === 62 ? 0 : 62));
    }, 2200);
    return () => clearInterval(interval);
  }, []);

  // Automatically fetch user GPS on load if permitted
  useEffect(() => {
    if ('geolocation' in navigator) {
      navigator.geolocation.getCurrentPosition(
        async (pos) => {
          const { latitude, longitude } = pos.coords;
          const calculated = calculateQiblaData(latitude, longitude);
          setQiblaData(calculated);

          try {
            const res = await fetch(
              `https://nominatim.openstreetmap.org/reverse?format=json&lat=${latitude}&lon=${longitude}`
            );
            const data = await res.json();
            const cityName =
              data.address?.city ||
              data.address?.town ||
              data.address?.suburb ||
              data.address?.state ||
              'Current Location';
            setLocation({ lat: latitude, lon: longitude, city: cityName });
          } catch {
            setLocation({
              lat: latitude,
              lon: longitude,
              city: `${latitude.toFixed(2)}°, ${longitude.toFixed(2)}°`,
            });
          }
        },
        () => {
          // Keep Dhaka default if GPS denied
        }
      );
    }
  }, []);

  // Setup Device Orientation / Compass Listeners
  useEffect(() => {
    if (
      typeof DeviceOrientationEvent !== 'undefined' &&
      typeof (DeviceOrientationEvent as any).requestPermission === 'function'
    ) {
      setNeedsPermission(true);
    }

    const handleOrientation = (event: any) => {
      let currentHeading: number | null = null;
      if (typeof event.webkitCompassHeading === 'number') {
        // iOS Safari
        currentHeading = event.webkitCompassHeading;
      } else if (typeof event.alpha === 'number') {
        // Android Chrome / Standard
        currentHeading = (360 - event.alpha) % 360;
      }

      if (currentHeading !== null && !isNaN(currentHeading)) {
        setHeading(currentHeading);
        setNeedsPermission(false);
      }
    };

    window.addEventListener('deviceorientationabsolute', handleOrientation, true);
    window.addEventListener('deviceorientation', handleOrientation, true);

    return () => {
      window.removeEventListener('deviceorientationabsolute', handleOrientation, true);
      window.removeEventListener('deviceorientation', handleOrientation, true);
    };
  }, []);

  // Trigger iOS Compass Permission
  const requestIOSCompass = async () => {
    try {
      const response = await (DeviceOrientationEvent as any).requestPermission();
      if (response === 'granted') {
        setNeedsPermission(false);
      }
    } catch (err) {
      console.error('Compass permission error:', err);
    }
  };

  // Manual GPS Refresh Button
  const handleUpdateGPS = () => {
    if (!('geolocation' in navigator)) return;
    navigator.geolocation.getCurrentPosition((pos) => {
      const { latitude, longitude } = pos.coords;
      setQiblaData(calculateQiblaData(latitude, longitude));
      setLocation({
        lat: latitude,
        lon: longitude,
        city: `${latitude.toFixed(2)}° N, ${longitude.toFixed(2)}° E`,
      });
    });
  };

  // Calculate relative turn angle when live compass is active
  let relativeAngle = 0;
  let isAligned = false;

  if (heading !== null) {
    relativeAngle = (qiblaData.bearing - heading + 360) % 360;
    if (relativeAngle > 180) relativeAngle -= 360;
    isAligned = Math.abs(relativeAngle) <= 4;

    if (isAligned && !hasBuzzedRef.current) {
      if (typeof navigator !== 'undefined' && navigator.vibrate) {
        navigator.vibrate(120);
      }
      hasBuzzedRef.current = true;
    } else if (!isAligned) {
      hasBuzzedRef.current = false;
    }
  }

  const cardinalDir = getCardinalDirection(qiblaData.bearing);

  return (
    <div className="min-h-screen bg-[#F7F5F0] text-[#1C1B18] py-8 px-4 sm:px-6 font-sans">
      <div className="max-w-4xl mx-auto">
        {/* Top Header */}
        <header className="mb-6">
          <h1 className="text-3xl font-bold tracking-tight text-[#1C1B18]">
            Qibla
          </h1>
          <p className="text-sm text-[#6E6A63] mt-0.5">{location.city}</p>
        </header>

        {/* Collapsible Tutorial Card (Matches 1st Image Top Banner) */}
        {showTutorial && (
          <section className="bg-white rounded-2xl border border-[#E6E2D8] shadow-sm p-6 sm:p-8 mb-10 transition-all">
            <h2 className="text-center text-lg font-bold text-[#1C1B18] mb-6">
              Finding the Qibla
            </h2>

            <div className="grid grid-cols-1 md:grid-cols-12 gap-8 items-center">
              {/* Left Illustration */}
              <div className="md:col-span-5 flex items-center justify-center gap-5">
                {/* Flat vs Vertical Posture Icons */}
                <div className="flex flex-col gap-7 items-center">
                  <div className="flex items-center gap-2.5">
                    <div className="w-11 h-2 bg-[#2B303A] rounded-full border-t-2 border-teal-400" />
                    <Check className="w-4 h-4 text-[#16A34A] stroke-[3]" />
                  </div>
                  <div className="flex items-center gap-2.5">
                    <div className="w-2 h-11 bg-gray-400 rounded-full" />
                    <X className="w-4 h-4 text-red-500 stroke-[3]" />
                  </div>
                </div>

                {/* Animated Phone Screen Mockup */}
                <div className="w-40 h-56 bg-[#F3EFE6] border-2 border-[#D5CFC2] rounded-3xl p-3 flex flex-col items-center justify-between shadow-inner">
                  <div className="w-9 h-1 bg-[#B8B2A6] rounded-full" />

                  <div className="w-28 h-28 bg-white rounded-full border border-[#D5CFC2] relative flex items-center justify-center">
                    <span className="absolute top-1 text-[9px] font-bold text-red-500">
                      N
                    </span>
                    <span className="absolute bottom-1 text-[9px] font-bold text-gray-400">
                      S
                    </span>
                    <span className="absolute right-1.5 text-[9px] font-bold text-gray-400">
                      E
                    </span>
                    <span className="absolute left-1.5 text-[9px] font-bold text-gray-400">
                      W
                    </span>

                    {/* Animated Gold Arrow inside Mockup */}
                    <div
                      className="w-full h-full absolute inset-0 transition-transform duration-1000 ease-in-out"
                      style={{ transform: `rotate(${mockDemoAngle}deg)` }}
                    >
                      <div className="absolute top-2.5 left-1/2 -translate-x-1/2 text-xs">
                        🕋
                      </div>
                      <div
                        className={`w-1 h-11 mx-auto mt-6 rounded-full transition-colors duration-500 ${
                          mockDemoAngle === 0 ? 'bg-[#16A34A]' : 'bg-[#8C6218]'
                        }`}
                      />
                    </div>
                  </div>

                  <span className="text-[11px] font-semibold text-[#1C1B18]">
                    {mockDemoAngle === 0
                      ? "You're facing Qibla"
                      : `Turn right ${mockDemoAngle}°`}
                  </span>
                </div>
              </div>

              {/* Right 3 Steps */}
              <div className="md:col-span-7 flex flex-col gap-4">
                <div className="flex items-start gap-3">
                  <span className="w-6 h-6 rounded-full bg-[#F3EDE2] text-[#8C6218] font-bold text-xs flex items-center justify-center shrink-0 mt-0.5">
                    1
                  </span>
                  <p className="text-sm text-[#3D3A35] leading-relaxed">
                    Hold your phone flat in front of you, screen up, like a tray.
                    Keep it away from metal, magnets and magnetic cases.
                  </p>
                </div>

                <div className="flex items-start gap-3">
                  <span className="w-6 h-6 rounded-full bg-[#F3EDE2] text-[#8C6218] font-bold text-xs flex items-center justify-center shrink-0 mt-0.5">
                    2
                  </span>
                  <p className="text-sm text-[#3D3A35] leading-relaxed">
                    The gold arrow always points to the Kaaba. Turn your whole
                    body slowly, the way the screen tells you.
                  </p>
                </div>

                <div className="flex items-start gap-3">
                  <span className="w-6 h-6 rounded-full bg-[#F3EDE2] text-[#8C6218] font-bold text-xs flex items-center justify-center shrink-0 mt-0.5">
                    3
                  </span>
                  <p className="text-sm text-[#3D3A35] leading-relaxed">
                    When the arrow points straight up to the top of your phone, it
                    turns green and the phone buzzes. You're facing the Qibla.
                  </p>
                </div>

                {/* Calibration Tip */}
                <div className="bg-[#F7F5F0] rounded-xl px-4 py-3 flex items-center gap-3 text-xs text-[#6E6A63] mt-1">
                  <span className="text-xl font-bold text-[#8C6218] leading-none">
                    ∞
                  </span>
                  <span>
                    Arrow jumping around? Step away from metal and move your
                    phone in a figure-8 a few times.
                  </span>
                </div>

                {/* Got It Button */}
                <div className="mt-2">
                  <button
                    onClick={() => setShowTutorial(false)}
                    className="bg-[#8C6218] hover:bg-[#734F12] text-white font-semibold text-sm px-14 py-3 rounded-xl transition-colors cursor-pointer"
                  >
                    Got it
                  </button>
                </div>
              </div>
            </div>
          </section>
        )}

        {/* Main 2-Column Compass & Metrics Area */}
        <main className="grid grid-cols-1 md:grid-cols-12 gap-10 items-center">
          {/* Left Column: 2D Interactive Compass Dial */}
          <div className="md:col-span-5 flex justify-center">
            <div className="w-72 h-72 bg-white rounded-full border border-[#DCD7CC] shadow-sm relative flex items-center justify-center">
              {/* Rotating Compass Ring */}
              <div
                className="w-full h-full relative transition-transform duration-150 ease-out"
                style={{
                  transform:
                    heading !== null ? `rotate(${-heading}deg)` : 'rotate(0deg)',
                }}
              >
                {/* Degree Ticks */}
                {Array.from({ length: 36 }).map((_, i) => (
                  <div
                    key={i}
                    className="absolute inset-0"
                    style={{ transform: `rotate(${i * 10}deg)` }}
                  >
                    <div
                      className={`mx-auto ${
                        i % 9 === 0
                          ? 'w-0.5 h-2.5 bg-gray-400'
                          : 'w-[1px] h-1.5 bg-gray-300'
                      }`}
                    />
                  </div>
                ))}

                {/* Cardinal Labels */}
                <span className="absolute top-3.5 left-1/2 -translate-x-1/2 text-xs font-bold text-red-600">
                  N
                </span>
                <span className="absolute bottom-3.5 left-1/2 -translate-x-1/2 text-xs font-bold text-[#6E6A63]">
                  S
                </span>
                <span className="absolute right-4 top-1/2 -translate-y-1/2 text-xs font-bold text-[#6E6A63]">
                  E
                </span>
                <span className="absolute left-4 top-1/2 -translate-y-1/2 text-xs font-bold text-[#6E6A63]">
                  W
                </span>

                {/* Dashed North Reference Line */}
                <div className="absolute top-8 bottom-1/2 left-1/2 -translate-x-1/2 border-l border-dashed border-gray-300" />

                {/* Qibla Pointer Arrow + Kaaba */}
                <div
                  className="w-full h-full absolute inset-0 transition-transform duration-200"
                  style={{ transform: `rotate(${qiblaData.bearing}deg)` }}
                >
                  {/* Kaaba Emoji/Icon */}
                  <div className="absolute top-5 left-1/2 -translate-x-1/2 text-lg select-none">
                    🕋
                  </div>

                  {/* Arrow Triangle Head */}
                  <div
                    className={`w-0 h-0 mx-auto mt-11 border-l-[12px] border-l-transparent border-r-[12px] border-r-transparent border-b-[20px] transition-colors duration-300 ${
                      isAligned ? 'border-b-[#16A34A]' : 'border-b-[#8C6218]'
                    }`}
                  />

                  {/* Arrow Shaft */}
                  <div
                    className={`w-2 h-20 mx-auto -mt-0.5 rounded-b-full transition-colors duration-300 ${
                      isAligned ? 'bg-[#16A34A]' : 'bg-[#8C6218]'
                    }`}
                  />
                </div>
              </div>

              {/* Center Pivot Dot */}
              <div
                className={`w-3.5 h-3.5 rounded-full z-10 transition-colors duration-300 ${
                  isAligned ? 'bg-[#16A34A]' : 'bg-[#8C6218]'
                }`}
              />
            </div>
          </div>

          {/* Right Column: Live Status, Controls & Cards */}
          <div className="md:col-span-7 text-center md:text-left">
            {/* Dynamic Status Heading */}
            {heading === null ? (
              <h2 className="text-xl sm:text-2xl font-bold text-[#1C1B18] leading-snug mb-4">
                Live compass is not available on this device. Face{' '}
                {qiblaData.bearing}° ({cardinalDir}) clockwise from true North.
              </h2>
            ) : isAligned ? (
              <h2 className="text-2xl font-bold text-[#16A34A] leading-snug mb-4">
                You're facing the Qibla
              </h2>
            ) : (
              <h2 className="text-2xl font-bold text-[#1C1B18] leading-snug mb-4">
                {relativeAngle > 0
                  ? `Turn right ${Math.round(Math.abs(relativeAngle))}°`
                  : `Turn left ${Math.round(Math.abs(relativeAngle))}°`}
              </h2>
            )}

            {/* Action Pills */}
            <div className="flex flex-wrap items-center justify-center md:justify-start gap-2.5 mb-4">
              <button
                onClick={() => setShowTutorial(true)}
                className="inline-flex items-center gap-1.5 bg-white hover:bg-gray-50 text-[#1C1B18] border border-[#E6E2D8] rounded-full px-4 py-1.5 text-xs font-semibold shadow-2xs cursor-pointer"
              >
                <Info className="w-3.5 h-3.5 text-[#8C6218]" />
                How to use
              </button>

              {needsPermission && (
                <button
                  onClick={requestIOSCompass}
                  className="inline-flex items-center gap-1.5 bg-[#8C6218] text-white rounded-full px-4 py-1.5 text-xs font-semibold shadow-2xs cursor-pointer"
                >
                  <Compass className="w-3.5 h-3.5" />
                  Enable Live Compass
                </button>
              )}

              <button
                onClick={handleUpdateGPS}
                className="inline-flex items-center gap-1.5 bg-white hover:bg-gray-50 text-[#1C1B18] border border-[#E6E2D8] rounded-full px-4 py-1.5 text-xs font-semibold shadow-2xs cursor-pointer"
              >
                <MapPin className="w-3.5 h-3.5 text-[#8C6218]" />
                Detect My GPS
              </button>
            </div>

            <p className="text-xs text-[#6E6A63] mb-6">
              {heading === null
                ? 'Open this page on your phone for a live compass, or use the angle above.'
                : 'Hold phone flat. The arrow turns green and buzzes when aligned with the Kaaba.'}
            </p>

            {/* Bottom 2 Metric Cards */}
            <div className="grid grid-cols-2 gap-4 text-left">
              <div className="bg-white rounded-xl border border-[#E6E2D8] p-4 sm:p-5">
                <div className="text-xs text-[#6E6A63] mb-1">
                  Qibla direction
                </div>
                <div className="text-2xl font-bold text-[#1C1B18]">
                  {qiblaData.bearing}°
                </div>
                <div className="text-[11px] text-[#6E6A63] mt-0.5">
                  {cardinalDir} · clockwise from true north
                </div>
              </div>

              <div className="bg-white rounded-xl border border-[#E6E2D8] p-4 sm:p-5">
                <div className="text-xs text-[#6E6A63] mb-1">
                  Distance to the Kaaba
                </div>
                <div className="text-2xl font-bold text-[#1C1B18]">
                  {qiblaData.distanceKm.toLocaleString()}
                </div>
                <div className="text-[11px] text-[#6E6A63] mt-0.5">
                  km · {qiblaData.distanceMi.toLocaleString()} mi
                </div>
              </div>
            </div>
          </div>
        </main>
      </div>
    </div>
  );
}

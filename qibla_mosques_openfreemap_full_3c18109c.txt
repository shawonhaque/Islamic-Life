// Kaaba Coordinates (Makkah)
const KAABA_LAT = 21.4225;
const KAABA_LON = 39.8262;

// Default Fallback Coordinates (Dhaka, Bangladesh)
const DEFAULT_COORDS = {
  lat: 23.7104,
  lon: 90.4074,
  city: 'Dhaka',
};

interface Mosque {
  id: number;
  name: string;
  address: string;
  lat: number;
  lon: number;
  distanceKm: number;
  phone?: string;
  jumuahTime?: string;
  verified: boolean;
}

function toRadians(deg: number) {
  return (deg * Math.PI) / 180;
}

function toDegrees(rad: number) {
  return (rad * 180) / Math.PI;
}

function calculateDistanceKm(lat1: number, lon1: number, lat2: number, lon2: number) {
  const R = 6371;
  const dLat = toRadians(lat2 - lat1);
  const dLon = toRadians(lon2 - lon1);
  const a =
    Math.sin(dLat / 2) * Math.sin(dLat / 2) +
    Math.cos(toRadians(lat1)) *
      Math.cos(toRadians(lat2)) *
      Math.sin(dLon / 2) *
      Math.sin(dLon / 2);
  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
  return R * c;
}

function calculateQiblaData(lat: number, lon: number) {
  const phi1 = toRadians(lat);
  const phi2 = toRadians(KAABA_LAT);
  const deltaLambda = toRadians(KAABA_LON - lon);

  const y = Math.sin(deltaLambda) * Math.cos(phi2);
  const x =
    Math.cos(phi1) * Math.sin(phi2) -
    Math.sin(phi1) * Math.cos(phi2) * Math.cos(deltaLambda);

  let bearing = toDegrees(Math.atan2(y, x));
  bearing = (bearing + 360) % 360;

  const distanceKm = Math.round(calculateDistanceKm(lat, lon, KAABA_LAT, KAABA_LON));
  const distanceMi = Math.round(distanceKm * 0.621371);

  return {
    bearing: Number(bearing.toFixed(1)),
    distanceKm,
    distanceMi,
  };
}

function getCardinalDirection(deg: number) {
  const dirs = ['N', 'NE', 'E', 'SE', 'S', 'SW', 'W', 'NW'];
  const index = Math.round(deg / 45) % 8;
  return dirs[index];
}

export default function App() {
  const [activeTab, setActiveTab] = useState<'mosques' | 'qibla'>('mosques');
  const [location, setLocation] = useState(DEFAULT_COORDS);

  // Qibla States
  const [showTutorial, setShowTutorial] = useState(true);
  const [qiblaData, setQiblaData] = useState(() =>
    calculateQiblaData(DEFAULT_COORDS.lat, DEFAULT_COORDS.lon)
  );
  const [heading, setHeading] = useState<number | null>(null);
  const [needsPermission, setNeedsPermission] = useState(false);
  const [mockDemoAngle, setMockDemoAngle] = useState(62);
  const hasBuzzedRef = useRef(false);

  // Mosques & OpenFreeMap States
  const [mosques, setMosques] = useState<Mosque[]>([]);
  const [loadingMosques, setLoadingMosques] = useState(false);
  const [searchQuery, setSearchQuery] = useState('');
  const [filterType, setFilterType] = useState<'all' | 'jumuah' | 'phone'>('all');
  const [selectedMosqueId, setSelectedMosqueId] = useState<number | null>(null);
  const [mapStyleMode, setMapStyleMode] = useState<'liberty' | 'positron' | 'earth'>('liberty');
  const [is3DBuildings, setIs3DBuildings] = useState(true);
  const [mapLoaded, setMapLoaded] = useState(false);

  const mapContainerRef = useRef<HTMLDivElement | null>(null);
  const mapInstanceRef = useRef<any>(null);
  const markersRef = useRef<any[]>([]);

  // Animate Tutorial Phone Arrow
  useEffect(() => {
    const interval = setInterval(() => {
      setMockDemoAngle((prev) => (prev === 62 ? 0 : 62));
    }, 2200);
    return () => clearInterval(interval);
  }, []);

  // Load MapLibre GL JS dynamically via CDN
  useEffect(() => {
    if (window.maplibregl) {
      setMapLoaded(true);
      return;
    }

    const link = document.createElement('link');
    link.rel = 'stylesheet';
    link.href = 'https://unpkg.com/maplibre-gl@4.7.1/dist/maplibre-gl.css';
    document.head.appendChild(link);

    const script = document.createElement('script');
    script.src = 'https://unpkg.com/maplibre-gl@4.7.1/dist/maplibre-gl.js';
    script.async = true;
    script.onload = () => setMapLoaded(true);
    document.head.appendChild(script);
  }, []);

  // Fetch Live GPS Location
  const detectUserGPS = () => {
    if (!('geolocation' in navigator)) return;
    navigator.geolocation.getCurrentPosition(
      async (pos) => {
        const { latitude, longitude } = pos.coords;
        setQiblaData(calculateQiblaData(latitude, longitude));

        try {
          const res = await fetch(
            `https://nominatim.openstreetmap.org/reverse?format=json&lat=${latitude}&lon=${longitude}`
          );
          const data = await res.json();
          const cityName =
            data.address?.city ||
            data.address?.town ||
            data.address?.suburb ||
            data.address?.state ||
            'Current Location';
          setLocation({ lat: latitude, lon: longitude, city: cityName });
        } catch {
          setLocation({ lat: latitude, lon: longitude, city: 'Current Location' });
        }
      },
      () => {
        // Keep Dhaka fallback if GPS denied
      }
    );
  };

  useEffect(() => {
    detectUserGPS();
  }, []);

  // Fetch Live Nearby Mosques from OpenStreetMap Overpass API
  useEffect(() => {
    const fetchMosques = async () => {
      setLoadingMosques(true);
      try {
        const radius = 3500; // 3.5km radius
        const query = `
          [out:json][timeout:25];
          (
            node["amenity"="place_of_worship"]["religion"="muslim"](around:${radius},${location.lat},${location.lon});
            way["amenity"="place_of_worship"]["religion"="muslim"](around:${radius},${location.lat},${location.lon});
          );
          out center;
        `;
        const res = await fetch('https://overpass-api.de/api/interpreter', {
          method: 'POST',
          body: query,
        });
        const data = await res.json();

        const parsed: Mosque[] = (data.elements || [])
          .map((el: any) => {
            const mLat = el.lat || el.center?.lat;
            const mLon = el.lon || el.center?.lon;
            if (!mLat || !mLon) return null;

            const tags = el.tags || {};
            const name =
              tags['name:en'] || tags.name || tags['name:bn'] || 'Local Jame Masjid';
            const street =
              tags['addr:full'] ||
              [tags['addr:housenumber'], tags['addr:street'], tags['addr:suburb']]
                .filter(Boolean)
                .join(', ') ||
              location.city;
            const phone = tags.phone || tags['contact:phone'] || undefined;
            const jumuahTime = tags['service_times:jumuah'] || tags['jumuah'] || undefined;
            const dist = calculateDistanceKm(location.lat, location.lon, mLat, mLon);

            return {
              id: el.id,
              name,
              address: street,
              lat: mLat,
              lon: mLon,
              distanceKm: Number(dist.toFixed(1)),
              phone,
              jumuahTime,
              verified: Boolean(tags.name && (tags['addr:street'] || phone)),
            };
          })
          .filter(Boolean)
          .sort((a: Mosque, b: Mosque) => a.distanceKm - b.distanceKm);

        setMosques(parsed);
      } catch (error) {
        console.error('Error fetching mosques:', error);
      } finally {
        setLoadingMosques(false);
      }
    };

    fetchMosques();
  }, [location.lat, location.lon]);

  // Get Map Style Configuration (OpenFreeMap & Earth Satellite)
  const getMapStyle = (mode: 'liberty' | 'positron' | 'earth') => {
    if (mode === 'liberty') {
      return 'https://tiles.openfreemap.org/styles/liberty';
    }
    if (mode === 'positron') {
      return 'https://tiles.openfreemap.org/styles/positron';
    }
    // Earth Satellite View with OpenFreeMap Attribution
    return {
      version: 8,
      sources: {
        satellite: {
          type: 'raster',
          tiles: [
            'https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/{z}/{y}/{x}',
          ],
          tileSize: 256,
          attribution:
            'OpenFreeMap © OpenMapTiles · Data from OpenStreetMap · Imagery © Esri',
        },
      },
      layers: [
        {
          id: 'satellite-layer',
          type: 'raster',
          source: 'satellite',
          minzoom: 0,
          maxzoom: 19,
        },
      ],
    };
  };

  // Initialize & Update MapLibre GL JS Map
  useEffect(() => {
    if (!mapLoaded || activeTab !== 'mosques' || !mapContainerRef.current) return;

    const maplibregl = window.maplibregl;
    if (!maplibregl) return;

    if (mapInstanceRef.current) {
      mapInstanceRef.current.remove();
      mapInstanceRef.current = null;
    }

    const map = new maplibregl.Map({
      container: mapContainerRef.current,
      style: getMapStyle(mapStyleMode),
      center: [location.lon, location.lat],
      zoom: 15,
      pitch: is3DBuildings ? 50 : 0,
      bearing: -15,
      antialias: true,
    });

    map.addControl(new maplibregl.NavigationControl(), 'top-right');

    map.on('style.load', () => {
      // Enable Google Earth-style 3D Globe projection when zoomed out
      if (map.setProjection) {
        map.setProjection({ type: 'globe' });
      }
    });

    // User Blue Pulsing Marker
    const userEl = document.createElement('div');
    userEl.innerHTML = `
      <div style="width:18px;height:18px;background:#2563EB;border:3px solid #FFFFFF;border-radius:50%;box-shadow:0 0 12px rgba(37,99,235,0.7);"></div>
    `;
    new maplibregl.Marker({ element: userEl })
      .setLngLat([location.lon, location.lat])
      .setPopup(new maplibregl.Popup({ offset: 15 }).setHTML('<b>Your Location</b>'))
      .addTo(map);

    mapInstanceRef.current = map;

    return () => {
      if (mapInstanceRef.current) {
        mapInstanceRef.current.remove();
        mapInstanceRef.current = null;
      }
    };
  }, [mapLoaded, activeTab, mapStyleMode, location.lat, location.lon]);

  // Toggle 3D Pitch smoothly
  useEffect(() => {
    if (mapInstanceRef.current) {
      mapInstanceRef.current.easeTo({
        pitch: is3DBuildings ? 50 : 0,
        duration: 600,
      });
    }
  }, [is3DBuildings]);

  // Plot Gold Mosque Pins on MapLibre
  useEffect(() => {
    const map = mapInstanceRef.current;
    const maplibregl = window.maplibregl;
    if (!map || !maplibregl) return;

    markersRef.current.forEach((m) => m.remove());
    markersRef.current = [];

    mosques.forEach((m) => {
      const el = document.createElement('div');
      const isSelected = m.id === selectedMosqueId;
      el.style.cursor = 'pointer';
      el.innerHTML = `
        <div style="
          width: ${isSelected ? '34px' : '28px'};
          height: ${isSelected ? '34px' : '28px'};
          background: ${isSelected ? '#8C6218' : '#FFFFFF'};
          border: 2.5px solid #8C6218;
          border-radius: 50%;
          display: flex;
          align-items: center;
          justify-content: center;
          box-shadow: 0 2px 6px rgba(0,0,0,0.2);
          font-size: 14px;
          transition: all 0.2s;
        ">
          🕌
        </div>
      `;

      const popupHtml = `
        <div style="font-family:sans-serif;padding:4px;max-width:210px;">
          <div style="font-weight:700;font-size:13px;color:#1C1B18;">${m.name}</div>
          <div style="font-size:11px;color:#6E6A63;margin-top:2px;">${m.address}</div>
          <div style="font-size:11px;font-weight:600;color:#8C6218;margin-top:4px;">${m.distanceKm} km away</div>
          <a href="https://www.google.com/maps/dir/?api=1&destination=${m.lat},${m.lon}" target="_blank" style="display:inline-block;margin-top:6px;font-size:11px;color:#2563EB;text-decoration:underline;">Get Directions →</a>
        </div>
      `;

      const marker = new maplibregl.Marker({ element: el })
        .setLngLat([m.lon, m.lat])
        .setPopup(new maplibregl.Popup({ offset: 18 }).setHTML(popupHtml))
        .addTo(map);

      el.addEventListener('click', () => {
        setSelectedMosqueId(m.id);
      });

      markersRef.current.push(marker);
    });
  }, [mosques, selectedMosqueId, mapStyleMode, activeTab]);

  // Fly to Mosque when clicked in Sidebar
  const handleSelectMosqueOnMap = (mosque: Mosque) => {
    setSelectedMosqueId(mosque.id);
    if (mapInstanceRef.current) {
      mapInstanceRef.current.flyTo({
        center: [mosque.lon, mosque.lat],
        zoom: 17,
        pitch: is3DBuildings ? 55 : 0,
        duration: 1200,
      });
    }
  };

  // Compass Orientation Sensor Listener
  useEffect(() => {
    if (
      typeof DeviceOrientationEvent !== 'undefined' &&
      typeof (DeviceOrientationEvent as any).requestPermission === 'function'
    ) {
      setNeedsPermission(true);
    }

    const handleOrientation = (event: any) => {
      let currentHeading: number | null = null;
      if (typeof event.webkitCompassHeading === 'number') {
        currentHeading = event.webkitCompassHeading;
      } else if (typeof event.alpha === 'number') {
        currentHeading = (360 - event.alpha) % 360;
      }

      if (currentHeading !== null && !isNaN(currentHeading)) {
        setHeading(currentHeading);
        setNeedsPermission(false);
      }
    };

    window.addEventListener('deviceorientationabsolute', handleOrientation, true);
    window.addEventListener('deviceorientation', handleOrientation, true);

    return () => {
      window.removeEventListener('deviceorientationabsolute', handleOrientation, true);
      window.removeEventListener('deviceorientation', handleOrientation, true);
    };
  }, []);

  const requestIOSCompass = async () => {
    try {
      const response = await (DeviceOrientationEvent as any).requestPermission();
      if (response === 'granted') {
        setNeedsPermission(false);
      }
    } catch (err) {
      console.error(err);
    }
  };

  // Qibla Alignment Calculation
  let relativeAngle = 0;
  let isAligned = false;

  if (heading !== null) {
    relativeAngle = (qiblaData.bearing - heading + 360) % 360;
    if (relativeAngle > 180) relativeAngle -= 360;
    isAligned = Math.abs(relativeAngle) <= 4;

    if (isAligned && !hasBuzzedRef.current) {
      if (typeof navigator !== 'undefined' && navigator.vibrate) {
        navigator.vibrate(120);
      }
      hasBuzzedRef.current = true;
    } else if (!isAligned) {
      hasBuzzedRef.current = false;
    }
  }

  // Filtered Mosques List
  const filteredMosques = mosques.filter((m) => {
    const matchesSearch =
      m.name.toLowerCase().includes(searchQuery.toLowerCase()) ||
      m.address.toLowerCase().includes(searchQuery.toLowerCase());
    if (!matchesSearch) return false;
    if (filterType === 'phone') return Boolean(m.phone);
    if (filterType === 'jumuah') return Boolean(m.jumuahTime);
    return true;
  });

  const cardinalDir = getCardinalDirection(qiblaData.bearing);

  return (
    <div className="min-h-screen bg-[#F7F5F0] text-[#1C1B18] flex flex-col font-sans">
      {/* Top Navigation Bar */}
      <nav className="bg-white border-b border-[#E6E2D8] px-4 sm:px-8 py-3 flex items-center justify-between sticky top-0 z-30">
        <div className="flex items-center bg-[#F7F5F0] p-1 rounded-xl border border-[#E6E2D8]">
          <button
            onClick={() => setActiveTab('mosques')}
            className={`px-4 py-1.5 rounded-lg text-xs font-semibold transition cursor-pointer flex items-center gap-1.5 ${
              activeTab === 'mosques'
                ? 'bg-[#8C6218] text-white shadow-2xs'
                : 'text-[#6E6A63] hover:text-[#1C1B18]'
            }`}
          >
            <MapPin className="w-3.5 h-3.5" />
            Live Mosques
          </button>
          <button
            onClick={() => setActiveTab('qibla')}
            className={`px-4 py-1.5 rounded-lg text-xs font-semibold transition cursor-pointer flex items-center gap-1.5 ${
              activeTab === 'qibla'
                ? 'bg-[#8C6218] text-white shadow-2xs'
                : 'text-[#6E6A63] hover:text-[#1C1B18]'
            }`}
          >
            <Compass className="w-3.5 h-3.5" />
            Qibla Compass
          </button>
        </div>

        <button
          onClick={detectUserGPS}
          className="inline-flex items-center gap-1.5 text-xs font-semibold bg-[#F7F5F0] hover:bg-[#EDE9DF] text-[#1C1B18] px-3 py-1.5 rounded-full border border-[#E6E2D8] cursor-pointer"
        >
          <LocateFixed className="w-3.5 h-3.5 text-[#8C6218]" />
          {location.city}
        </button>
      </nav>

      {/* TAB 1: LIVE MOSQUES FINDER (OpenFreeMap + Earth 3D View) */}
      {activeTab === 'mosques' && (
        <div className="flex-1 flex flex-col md:flex-row h-[calc(100vh-57px)] overflow-hidden">
          {/* Left Sidebar */}
          <aside className="w-full md:w-[410px] bg-[#F7F5F0] border-r border-[#E6E2D8] flex flex-col h-1/2 md:h-full">
            <div className="p-5 pb-3">
              <h1 className="text-2xl font-bold text-[#1C1B18]">Mosques</h1>
              <p className="text-xs text-[#6E6A63] mt-0.5">
                Near {location.city} · {filteredMosques.length} mosques
              </p>

              {/* Search Bar */}
              <div className="relative mt-3.5">
                <Search className="w-4 h-4 text-[#8C877E] absolute left-3.5 top-1/2 -translate-y-1/2" />
                <input
                  type="text"
                  value={searchQuery}
                  onChange={(e) => setSearchQuery(e.target.value)}
                  placeholder="Search mosques by name or street"
                  className="w-full bg-white border border-[#E6E2D8] rounded-xl pl-10 pr-4 py-2.5 text-xs text-[#1C1B18] placeholder-[#8C877E] focus:outline-none focus:border-[#8C6218]"
                />
              </div>

              {/* Filter Pills */}
              <div className="flex items-center gap-2 mt-3 overflow-x-auto pb-1">
                <button
                  onClick={() => setFilterType('all')}
                  className={`px-3 py-1.5 rounded-full text-xs font-semibold whitespace-nowrap cursor-pointer transition ${
                    filterType === 'all'
                      ? 'bg-[#8C6218] text-white'
                      : 'bg-white border border-[#E6E2D8] text-[#4A463F]'
                  }`}
                >
                  🕌 All
                </button>
                <button
                  onClick={() => setFilterType('jumuah')}
                  className={`px-3 py-1.5 rounded-full text-xs font-semibold whitespace-nowrap cursor-pointer transition flex items-center gap-1 ${
                    filterType === 'jumuah'
                      ? 'bg-[#8C6218] text-white'
                      : 'bg-white border border-[#E6E2D8] text-[#4A463F]'
                  }`}
                >
                  <Clock className="w-3 h-3" /> Has Jumu'ah time
                </button>
                <button
                  onClick={() => setFilterType('phone')}
                  className={`px-3 py-1.5 rounded-full text-xs font-semibold whitespace-nowrap cursor-pointer transition flex items-center gap-1 ${
                    filterType === 'phone'
                      ? 'bg-[#8C6218] text-white'
                      : 'bg-white border border-[#E6E2D8] text-[#4A463F]'
                  }`}
                >
                  <Phone className="w-3 h-3" /> Has phone
                </button>
              </div>
            </div>

            {/* Scrollable Mosque Cards List */}
            <div className="flex-1 overflow-y-auto px-5 pb-6 space-y-3">
              {loadingMosques ? (
                <div className="text-center py-10 text-xs text-[#6E6A63]">
                  Finding nearby mosques from OpenStreetMap...
                </div>
              ) : filteredMosques.length === 0 ? (
                <div className="text-center py-10 text-xs text-[#6E6A63]">
                  No mosques found matching your filter.
                </div>
              ) : (
                filteredMosques.map((mosque) => (
                  <div
                    key={mosque.id}
                    onClick={() => handleSelectMosqueOnMap(mosque)}
                    className={`bg-white rounded-xl p-4 border transition cursor-pointer ${
                      selectedMosqueId === mosque.id
                        ? 'border-[#8C6218] shadow-sm'
                        : 'border-[#E6E2D8] hover:border-[#C9C2B4]'
                    }`}
                  >
                    <div className="flex items-start justify-between gap-2">
                      <div>
                        <h3 className="font-bold text-sm text-[#1C1B18] leading-snug">
                          {mosque.name}
                        </h3>
                        <p className="text-xs text-[#6E6A63] mt-0.5">
                          {mosque.address}
                        </p>
                      </div>
                      <span className="bg-[#F3EDE2] text-[#8C6218] text-[11px] font-bold px-2.5 py-0.5 rounded-full shrink-0">
                        {mosque.distanceKm} km
                      </span>
                    </div>

                    <p className="text-[11px] text-[#8C877E] mt-2">
                      Jumu'ah{' '}
                      <span className="italic">
                        {mosque.jumuahTime || 'Not added yet'}
                      </span>
                    </p>

                    {/* Card Action Buttons */}
                    <div className="flex items-center justify-between mt-3 pt-2.5 border-t border-[#F3EFE6]">
                      <div className="flex items-center gap-2">

                            onClick={(e) => e.stopPropagation()}
                            className="inline-flex items-center gap-1 bg-[#F7F5F0] hover:bg-[#EFECE4] text-[#3D3A35] text-[11px] font-semibold px-2.5 py-1.5 rounded-lg"

                            handleSelectMosqueOnMap(mosque);
